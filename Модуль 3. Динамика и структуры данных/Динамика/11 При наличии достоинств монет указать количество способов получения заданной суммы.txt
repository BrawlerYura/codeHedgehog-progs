#include <iostream>
#include <vector>

using namespace std;

int main() {
    int s, n;
    cin >> n >> s;
    vector<int> coinsValue(n);
    for (int i = 0; i < n; i++) {
        cin >> coinsValue[i];
    }

    vector<int> arr(s + 1);
    arr[0] = 1;
    for (int i = 0; i < n; i++) {
        for (int j = 1; j < s + 1; j++) {
            if (j >= coinsValue[i]) {
                arr[j] += arr[j - coinsValue[i]];
                arr[j] %= 1000000007;
            }
        }
    }
    cout << arr[s];
}
