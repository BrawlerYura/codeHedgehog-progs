#include <iostream>
#include <stdio.h>

int max(int first, int second)
{
    if (first >= second) {
        return first;
    }
    return second;
}

int coordinate_y = 0;
int coordinate_x = 0;
int check(int a, int b)
{
    if ((a < coordinate_y) && (b < coordinate_x) && (a >= 0) && (b >= 0)) {
        return 1;
    }
    return 0;
}

int dynamics(int** arr, int** arrmax, int indy, int indx)
{
    if ((indy == 0) && (indx == 0)) {
        arrmax[0][0] = arr[0][0];
        return arr[0][0];
    }
    int first = -100000000;
    int second = -100000000;
    int third = -100000000;
    int fourth = -100000000;
    if (check(indy - 1, indx - 2) == 1) {
        if (arrmax[indy - 1][indx - 2] == -100000000) {
            first = dynamics(arr, arrmax, indy - 1, indx - 2);
            arrmax[indy - 1][indx - 2] = first;
        } else {
            first = arrmax[indy - 1][indx - 2];
        }
    }
    if (check(indy + 1, indx - 2) == 1) {
        if (arrmax[indy + 1][indx - 2] == -100000000) {
            second = dynamics(arr, arrmax, indy + 1, indx - 2);
            arrmax[indy + 1][indx - 2] = second;
        } else {
            second = arrmax[indy + 1][indx - 2];
        }
    }
    if (check(indy - 2, indx - 1) == 1) {
        if (arrmax[indy - 2][indx - 1] == -100000000) {
            third = dynamics(arr, arrmax, indy - 2, indx - 1);
            arrmax[indy - 2][indx - 1] = third;
        } else {
            third = arrmax[indy - 2][indx - 1];
        }
    }
    if (check(indy - 2, indx + 1) == 1) {
        if (arrmax[indy - 2][indx + 1] == -100000000) {
            fourth = dynamics(arr, arrmax, indy - 2, indx + 1);
            arrmax[indy - 2][indx + 1] = fourth;
        } else {
            fourth = arrmax[indy - 2][indx + 1];
        }
    }
    if ((first == -100000000) && (second == -100000000) && (third == -100000000) && (fourth == -100000000)) {
        return -100000000;
    } else {
        return max(max(first, second), max(third, fourth)) + arr[indy][indx];
    }
}

int main()
{
    scanf("%d", &coordinate_y);
    scanf("%d", &coordinate_x);

    int** glade = new int* [coordinate_y];
    for (int index_y = 0; index_y < coordinate_y; ++index_y) {
        glade[index_y] = new int[coordinate_x];
    }

    int** glade_max = new int* [coordinate_y];
    for (int index_y = 0; index_y < coordinate_y; ++index_y)  {
        glade_max[index_y] = new int[coordinate_x];
    }

    for (int index_y = 0; index_y < coordinate_y; ++index_y) {
        for (int index_x = 0; index_x < coordinate_x; ++index_x) {
            scanf("%d", &glade[index_y][index_x]);
            glade_max[index_y][index_x] = -100000000;
        }
    }

    int answer = dynamics(glade, glade_max, coordinate_y - 1, coordinate_x - 1);

    if (answer == -100000000) {
        printf("-");
    } else {
        printf("%d", answer);
    }
    return 0;
}
