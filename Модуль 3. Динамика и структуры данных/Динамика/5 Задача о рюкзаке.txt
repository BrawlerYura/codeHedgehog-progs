#include <iostream>

using namespace std;

struct Item {
    int weight;
};

int main() {
    int maxCapacity;
    int itemsLength;

    cin >> itemsLength >> maxCapacity;

    Item *items = new Item[itemsLength];
    for (int i = 0; i < itemsLength; i++) {
        cin >> items[i].weight;
    }

    int **arr = new int *[itemsLength + 1];
    for (int i = 0; i < itemsLength + 1; i++) {
        arr[i] = new int[maxCapacity + 1];
    }

    for (int i = 0; i < itemsLength + 1; i++) {
        for (int j = 0; j < maxCapacity + 1; j++) {
            if (i == 0 || j == 0) {
                arr[i][j] = 0;
            } else {
                if (items[i - 1].weight > j) {
                    arr[i][j] = arr[i - 1][j];
                } else {
                    int pre = arr[i - 1][j];
                    int alterItem = items[i - 1].weight + arr[i - 1][j - items[i - 1].weight];

                    arr[i][j] = max(pre, alterItem);
                }
            }
        }
    }

    cout << arr[itemsLength][maxCapacity];
}
