#include <iostream>

using namespace std;

void turtle(int **lawn, int n, int m) {
    for (int i = n - 2; i >= 0; i--) {
        lawn[i][m - 1] += lawn[i + 1][m - 1];
    }
    for (int j = m - 2; j >= 0; j--) {
        lawn[n - 1][j] += lawn[n - 1][j + 1];
    }
    for (int i = n - 2; i >= 0; i--) {
        for (int j = m - 2; j >= 0; j--) {
            lawn[i][j] += min(lawn[i + 1][j], lawn[i][j + 1]);
        }
    }
    cout << lawn[0][0];
}

int main() {
    int n, m;
    cin >> n >> m;
    int **lawn = new int *[n];
    for (int i = 0; i < n; i++) {
        lawn[i] = new int[m];
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> lawn[i][j];
        }
    }
    turtle(lawn, n, m);
}
