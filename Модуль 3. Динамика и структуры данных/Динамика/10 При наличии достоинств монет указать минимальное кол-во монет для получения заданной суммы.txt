#include <iostream>
#include <vector>
#include <climits>

using namespace std;

int main() {
    int n, s;
    cin >> n >> s;
    vector<int> nominalValue(n);
    for (int i = 0; i < n; i++) {
        cin >> nominalValue[i];
    }
    vector<int> arr(s + 1, INT_MAX);
    arr[0] = 0;
    for (int i = 1; i <= s; i++) {
        for (int j = 0; j < n; j++) {
            if (i - nominalValue[j] >= 0 && arr[i - nominalValue[j]] != INT_MAX) {
                arr[i] = min(arr[i], arr[i - nominalValue[j]] + 1);
            }
        }
    }
    if (arr[s] == INT_MAX) {
        cout << -1 << endl;
    } else {
        cout << arr[s] << endl;
    }
}
