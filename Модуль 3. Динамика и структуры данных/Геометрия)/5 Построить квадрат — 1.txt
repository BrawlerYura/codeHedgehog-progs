#include <iostream>
#include <cmath>

using namespace std;

struct Point {
    int x, y;
};

int main() {
    Point a, b;
    cin >> a.x >> a.y >> b.x >> b.y;
    int deltaX = a.x - b.x;
    int deltaY = a.y - b.y;
    Point c, d;
    c.x = deltaY + b.x;
    c.y = -deltaX + b.y;
    d.x = deltaY + a.x;
    d.y = -deltaX + a.y;
    cout << c.x << " " << c.y << " " << d.x << " " << d.y;
}
