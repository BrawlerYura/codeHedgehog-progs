#include <iostream>
#include <cmath>

using namespace std;
const double PI = 3.141592653589793;

struct Point {
    double x, y;
};

Point findPoint(Point a, Point b, double d) {
    Point c;
    c.x = (a.x * cos(d) - a.y * sin(d)) + b.x;
    c.y = (a.x * sin(d) + a.y * cos(d)) + b.y;
    return c;
}

int main() {
    double lengthAB, lengthAC, angleA;
    cin >> lengthAB >> lengthAC >> angleA;
    Point a, b, c;
    a.x = 0;
    a.y = 0;
    b.x = lengthAB;
    b.y = 0;
    c.x = lengthAC;
    c.y = 0;
    c = findPoint(c, a, angleA * PI / 180);
    printf("%.10f %.10f\n%.10f %.10f\n%.10f %.10f\n", a.x, a.y, b.x, b.y, c.x, c.y);
}
