#include <iostream>
#include <algorithm>

using namespace std;

struct Queue {
    int* arr;
    int top;
    int cashiers;
    int current;

    int serviceTime();

    void push(int x);

    int pop();

    Queue(int size, int cashiers) {
        arr = new int[size];
        top = -1;
        current = -1;
        this->cashiers = cashiers;
    }
};

void Queue::push(int x) {
    arr[++top] = x;
}

int Queue::pop() {
    current++;
    return arr[current];
}

void Heapify(int *arr, int n, int i)
{
    int largest = i;
    int l = 2*i + 1;
    int r = 2*i + 2;
    if (l < n && arr[l] > arr[largest])
        largest = l;
    if (r < n && arr[r] > arr[largest])
        largest = r;
    if (largest != i)
    {
        swap(arr[i], arr[largest]);
        Heapify(arr, n, largest);
    }
}

void heapSort(int *arr, int n)
{
    for (int i = n / 2 - 1; i >= 0; i--)
        Heapify(arr, n, i);
    for (int i=n-1; i>=0; i--)
    {
        swap(arr[0], arr[i]);
        Heapify(arr, i, 0);
    }
}

void heapify(int* arr, int n, int m)
{
    int largest = n;
    int k = 2 * n + 1;
    while (k <= m) {
        if (k < m && arr[k] > arr[k + 1]) {
            k++;
        }
        if (arr[largest] > arr[k]) {
            swap(arr[largest], arr[k]);
            largest = k;
            k = k * 2 + 1;
        }
        else
            break;
    }
}

int Queue::serviceTime() {
    int countTime = 0, max = -100000, min = 100000, sum = 0;
    int* cashiersArr = new int[cashiers];
    if(cashiers >= top + 1) {
        for (int i = 0; i < top + 1; i++) {
            cashiersArr[i] = arr[i];
        }
        for(int k = 0; k < cashiers; k++) {
            if(max < cashiersArr[k]) {
                max = cashiersArr[k];
            }
        }
        return max;
    } else if(cashiers == 1) {
        for(int j = 0; j <= top; j++) {
            sum += arr[j];
        }
        return sum;
    }
    for (int i = 0; i < cashiers; i++) {
        cashiersArr[i] = pop();
    }
    heapSort(cashiersArr, cashiers);
    for(int i = 0; i < top; i++) {
        if(cashiers + i < top + 1) {
            heapify(cashiersArr, 0, cashiers - 1);
            countTime += cashiersArr[0];
            for(int j = 1; j < cashiers; j++) {
                cashiersArr[j] -= cashiersArr[0];
            }
            cashiersArr[0] = pop();
        } else {
            for(int k = 0; k < cashiers; k++) {
                if(max < cashiersArr[k]) {
                    max = cashiersArr[k];
                }
            }
            break;
        }
    }
    countTime += max;
    return countTime;
}

int main() {
    int size;
    cin >> size;
    int cashiers;
    cin >> cashiers;
    Queue queue(size, cashiers);
    int time;
    for (int i = 0; i < size; i++) {
        cin >> time;
        queue.push(time);
    }
    cout << queue.serviceTime();
}
