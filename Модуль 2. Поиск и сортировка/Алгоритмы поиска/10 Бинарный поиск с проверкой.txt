#include <iostream>

using namespace std;

bool check(int *arr[], int n, double interval, double minDistance) {
    double firstCoordinate = 0, secondCoordinate = 0, firstTrainTime = arr[0][0], secondTrainTime = arr[0][0];
    int firstIndex = 0, secondIndex = 0;

    while (firstIndex < n) {
        if (interval > firstTrainTime) {
            firstCoordinate += arr[firstIndex][0] * arr[firstIndex][1];
            interval -= arr[firstIndex][0];
            firstIndex++;
            if (firstIndex > n - 1) {
                break;
            }
            firstTrainTime = arr[firstIndex][0];
        } else {
            firstCoordinate += arr[firstIndex][1] * interval;
            firstTrainTime = arr[firstIndex][0] - interval;
            break;
        }
    }

    if (firstCoordinate < minDistance) {
        return false;
    } else {
        while (firstIndex < n) {
            if (firstTrainTime > secondTrainTime) {
                firstCoordinate += arr[firstIndex][1] * secondTrainTime;
                secondCoordinate += arr[secondIndex][1] * secondTrainTime;

                firstTrainTime -= secondTrainTime;
                secondIndex++;
                secondTrainTime = arr[secondIndex][0];
            } else {
                if (secondTrainTime > firstTrainTime) {
                    firstCoordinate += arr[firstIndex][1] * firstTrainTime;
                    secondCoordinate += arr[secondIndex][1] * firstTrainTime;

                    secondTrainTime -= firstTrainTime;
                    firstIndex++;
                    if (firstIndex > n - 1) {
                        break;
                    }
                    firstTrainTime = arr[firstIndex][0];
                } else {
                    firstCoordinate += arr[firstIndex][1] * firstTrainTime;
                    secondCoordinate += arr[secondIndex][1] * secondTrainTime;

                    firstIndex++;
                    if (firstIndex > n - 1) {
                        break;
                    }
                    firstTrainTime = arr[firstIndex][0];
                    secondIndex++;
                    secondTrainTime = arr[secondIndex][0];
                }
            }

            if (firstCoordinate - secondCoordinate < minDistance) {
                return false;
            }

        }
    }

    if (firstCoordinate - secondCoordinate < minDistance) {
        return false;
    } else {
        return true;
    }
}

int main() {
    double left = 0, right = 0, mid, minDistance;
    int n;
    cin >> minDistance >> n;

    int **arr = new int *[n];
    for (int i = 0; i < n; i++) {
        arr[i] = new int[2];
        cin >> arr[i][0];
        cin >> arr[i][1];
        right += arr[i][1];
    }

    while (right - left + 0.001 > 0) {
        mid = (right + left) / 2;
        if (check(arr, n, mid, minDistance)) {
            right = mid - 0.001;
        } else {
            left = mid + 0.001;
        }
    }
    printf("%.3f", mid);
}
