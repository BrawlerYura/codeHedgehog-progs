#include <iostream>

using namespace std;
int Shift(int l, int r, int v1, int v2) {
    int mid, v;
    while (r - l > 1) {
        mid = (l + r) / 2;
        cout << "? " << mid << endl;
        cout.flush();
        cin >> v;
        if (v > v1 && v > v2) {
            l = mid;
            v1 = v;
        }
        else if (v < v1 && v < v2) {
            r = mid;
            v2 = v;
        }
        else if (v == v1 || v == v2)
            break;
    }
    return r;
}

void BinSearch(int l, int r, int rv) {
    int v, mid;
    bool a = true;
    while (l <= r) {
        mid = (l + r) / 2;
        cout << "? " << mid << endl;
        cout.flush();
        cin >> v;
        if (v == rv) {
            cout << "! " << mid << endl;
            cout.flush();
            a = false;
            break;
        }
        else if (v < rv)
            l = mid + 1;
        else
            r = mid - 1;
    }
    if (a) {
        cout << "! -1" << endl;
        cout.flush();
    }
}

int main() {
    int n, m, r, l, v1, v2, sheesh, rv;
    cin >> n >> m >> rv;
    cout << "? 1" << endl;
    cout.flush();
    cin >> v1;
    cout << "? " << n << endl;
    cout.flush();
    cin >> v2;
    r = n;
    l = 1;
    if (v1 > v2) {
        sheesh = Shift(l, r, v1, v2);
    }
    else {
        for (int i = 0; i < m; i++) {
            if (i != 0) {
                cin >> rv;
                BinSearch(l, r, rv);
            }
            else
                BinSearch(l, r, rv);
        }
        return 0;
    }
    for (int i = 0; i < m; i++) {
        if (i != 0) {
            cin >> rv;
            if (v1 == rv) {
                cout << "! 1" << endl;
                cout.flush();
            }
            else if (v2 == rv) {
                cout << "! " << n << endl;
                cout.flush();
            }
            else if (v1 < rv) 
                BinSearch(2, sheesh - 1, rv);
            else
                BinSearch(sheesh, n, rv);
        }
        else {
            if (v1 == rv) {
                cout << "! 1" << endl;
                cout.flush();
            }
            else if (v2 == rv) {
                cout << "! " << n << endl;
                cout.flush();
            }
            else if (v1 < rv) {
                BinSearch(2, sheesh - 1, rv);
            }
            else
                BinSearch(sheesh, n, rv);
        }
    }
}
