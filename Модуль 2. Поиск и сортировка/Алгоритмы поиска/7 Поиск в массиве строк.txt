#include <iostream>

using namespace std;
int Compare(string a,string b){
    for(int i = 0; i < min(a.length(), b.length()); i++){
        if(a[i] < b[i])
            return 2;
        else if(a[i] > b[i])
            return 3;
    }
    if(a.length() == b.length())
        return 1;
    else if(a.length() < b.length())
        return 2;
    else if(a.length() > b.length())
        return 0;
}
int main() {
    int n, m, r, l, mid;
    bool find;
    cin >> n;
    string *arr = new string[n];
    string word;
    for(int i = 0; i < n; i++)
        cin >> arr[i];
    cin >> m;
    for(int i = 0; i < m; i++){
        cin >> word;
        find = false;
        r = n - 1;
        l = 0;
        while (l <= r) {
            mid = (r + l) / 2;
            if(Compare(word, arr[mid]) == 1){
                find = true;
                cout << mid + 1 << endl;
                break;
            }
            else if(Compare(word, arr[mid]) == 0)
                l = mid + 1;
            else if(Compare(word, arr[mid]) == 2)
                r = mid - 1;
            else
                l = mid + 1;
        }
        if(!find)
            cout << "-1" << endl;
    }
}
