#include <iostream>

using namespace std;

int main() {
    int n, v;
    int r, l = 2, mid;
    cin >> n;
    r = n - 1;
    cout << "? 1" << endl;
    cout.flush();
    cin >> v;
    if(v == 1){
        cout << "! " << n << endl;
        cout.flush();
        return 0;
    }
    cout << "? " << n << endl;
    cout.flush();
    cin >> v;
    if(v == 0){
        cout << "! 0" << endl;
        cout.flush();
        return 0;
    }
    
    while(true){
        mid = (r + l) / 2;
        cout << "? " << mid << endl;
        cout.flush();
        cin >> v;
        if (v == 1) {
            cout << "? " << mid - 1 << endl;
            cout.flush();
            cin >> v;
            if(v == 0){
                break;
            }
            r = mid - 1;
        }
        else {
            cout << "? " << mid + 1 << endl;
            cout.flush();
            cin >> v;
            if(v == 1){
                break;
            }
            l = mid + 1;
        }
    }
    if(v == 0) {
        cout << "! " << n - mid + 1<< endl;
        cout.flush();
    }
    else {
        cout << "! " << n - mid << endl;
        cout.flush();
    }
}
