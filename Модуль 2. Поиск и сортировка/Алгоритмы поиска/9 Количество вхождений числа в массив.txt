#include <iostream>
#include <string>

using namespace std;

int main() {
    int r, l, n, m, mid, vl, vr;
    long long v;
    bool a = false;
    cin >> n;
    long long *arr = new long long[n];
    for(int i = 0; i < n; i++){
        cin >> arr[i];
    }
    cin >> m;
    for(int i = 0; i < m; i++){
        cin >> v;
        a = false;
        r = n;
        l = 0;
        while(l <= r){
            mid = (l+r)/2;
            if(v == arr[mid]){
                if(v != arr[mid - 1]){
                    vl = mid;
                    a = true;
                    break;
                }
                else{
                    r = mid - 1;
                }
            }
            else if(v > arr[mid]){
                l = mid + 1;
            }
            else{
                r = mid - 1;
            }
        }
        r = n;
        l = 0;
        while(l <= r){
            mid = (l+r)/2;
            if(v == arr[mid]){
                if(v != arr[mid + 1]){
                    vr = mid;
                    a = true;
                    break;
                }
                else{
                    l = mid + 1;
                }
            }
            else if(v > arr[mid]){
                l = mid + 1;
            }
            else{
                r = mid - 1;
            }
        }
        if(a){
            cout << vr - vl + 1 << endl;
        }
        else{
            cout << "Not found" << endl;
        }
    }
}
