#include <iostream>

using namespace std;

void heapify(int n, int i, int *solved, int *time, int *index)
{
    int largest = i;
    int l = 2*i + 1;
    int r = 2*i + 2;
    if (l < n && (solved[l] < solved[largest] || (solved[l] == solved[largest] && time[l] > time[largest]) || (solved[l] == solved[largest] && time[l] == time[largest] && index[l] > index[largest])))
        largest = l;
    if (r < n && (solved[r] < solved[largest] || (solved[r] == solved[largest] && time[r] > time[largest]) || (solved[r] == solved[largest] && time[r] == time[largest] && index[r] > index[largest])))
        largest = r;
    if (largest != i)
    {
        swap(solved[i], solved[largest]);
        swap(time[i], time[largest]);
        swap(index[i], index[largest]);
        heapify(n, largest, solved, time, index);
    }
}

void HeapSort(int n, int *solved, int *time, int *index)
{
    for (int i = n / 2 - 1; i >= 0; i--)
        heapify(n, i, solved, time, index);
    for (int i=n-1; i>=0; i--)
    {
        swap(solved[0], solved[i]);
        swap(time[0], time[i]);
        swap(index[0], index[i]);
        heapify(i, 0, solved, time, index);
    }
}

int main() {
    int n;
    scanf("%d", &n);
    int solved[n];
    int time[n];
    int index[n];
    for(int i = 0; i < n; i++){
        scanf("%d", &solved[i]);
        scanf("%d", &time[i]);
        index[i] = i + 1;
    }
    HeapSort(n, solved, time, index);

    for(int i = 0; i < n; i++)
        printf("%d ", index[i]);
}
