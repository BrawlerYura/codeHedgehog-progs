#define _CRT_SECURE_NO_WARNINGS
#include <iostream>
using namespace std;
void heapify(int* arr, int n, int m)
{
    int largest = n;
    int k = 2 * n + 1;
    while (k <= m) {
        if (k < m && arr[k] < arr[k + 1]) {
            k++;
        }
        if (arr[largest] < arr[k]) {
            swap(arr[largest], arr[k]);
            largest = k;
            k = k * 2 + 1;
        }
        else
            break;
    }
}

void HeapSort(int* arr, int n)
{
    for (int i = n / 2; i >= 0; i--)
        heapify(arr, i, n - 1);
    for (int i = n - 1; i >= 1; i--)
    {
        swap(arr[0], arr[i]);
        heapify(arr, 0, i - 1);
    }
}




int main()
{
    int n, k;
    int temp;
    scanf("%d", &n);
    scanf("%d", &k);
    int* arr = new int[k];
    for (int i = 0; i < k; i++) {
        scanf("%d", &temp);
        arr[i] = temp;
    }
    for (int i = k / 2; i >= 0; i--)
        heapify(arr, i, k - 1);
    for (int i = 0; i < n - k; i++) {
        scanf("%d", &temp);
        if (temp < arr[0]) {
            arr[0] = temp;
            heapify(arr, 0, k - 1);
        }
    }
    printf("%d\n", arr[0]);
}
