#include <iostream>
#include <cstdlib>
using namespace std;
void quickSort(int *arr, int l, int r)
{
    int i = l;
    int j = r;
    int mid = arr[(l + r) / 2 + 1];
    while (i <= j)
    {
        while (arr[i] < mid)
            i++;
        while (arr[j] > mid)
            j--;
        if (i <= j)
        {
            swap(arr[i], arr[j]);
            i++;
            j--;
        }
    }
    if (j > l)
        quickSort(arr, l, j);
    if (i < r)
        quickSort(arr, i, r);
}

int main()
{
    int n;
    cin >> n;
    int *arr = new int[n];
    for(int i = 0; i < n; i++)
        cin >> arr[i];
    QuickSort(arr, 0, n-1);
    for(int i = 0; i < n; i++)
        cout << arr[i] << " ";
}
