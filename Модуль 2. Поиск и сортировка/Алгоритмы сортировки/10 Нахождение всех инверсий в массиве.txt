#include <iostream>
using namespace std;
long long n, inversions = 0;
void Merge(long long l, long long r, long long *arr){
    long long mid, start, final;
    long long *temp = new long long[r - l + 1];
    mid = (l + r) / 2;
    start = l;
    final = mid + 1;
    for(int i = 0; i < (r - l) + 1; i++){
    if(arr[start] > arr[final]){
    temp[i] = arr[final];
    final++;
    inversions += mid + 1 - start;
    }
    else{
        temp[i] = arr[start];
        start++;
    }
    if(start == mid + 1){
        long long j = i + 1;
        while(final <= r + 1 && j < (r - l) + 1){
            temp[j] = arr[final];
            j++;
            final++;
            inversions += mid + 1 - start;
        }
        break;
    }
    if(final == r + 1){
        long long j = i + 1;
        while(start <= mid + 1 && j < (r - l) + 1){
            temp[j] = arr[start];
            j++;
            start++;
        }
        break;
    }
    }
    for(long long i = l; i <= r;i++)
        arr[i] = temp[i - l];
}
void MergeSort(long long l, long long r, long long *arr){
    if(l < r){
        MergeSort(l, (l + r) / 2, arr);
        MergeSort((l + r) / 2 + 1, r, arr);
        Merge(l, r, arr);
    }
}

int main()
{
    cin >> n;
    long long *arr = new long long[n];
    for(int i = 0; i < n; i++)
        cin >> arr[i];
    MergeSort(0, n - 1, arr);
    cout << inversions;
}
