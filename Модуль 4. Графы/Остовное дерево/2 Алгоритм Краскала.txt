#include <iostream>
#include <vector>
#include <climits>
#include <algorithm>

using namespace std;

struct Edges {
    int begin, end, weight;
};

bool comp (Edges a, Edges b) {
    return a.weight < b.weight;
}

void kraskalsAlg(vector<Edges> &edges, int n, vector<vector<int>>& matrix) {
    vector<bool> isUsed(n, false);
    int length = 0;
    matrix[edges[0].begin][edges[0].end] = edges[0].weight;
    matrix[edges[0].end][edges[0].begin] = edges[0].weight;
    isUsed[edges[0].begin] = true;
    isUsed[edges[0].end] = true;
    length += edges[0].weight;
    for(int i = 0; i < n - 2; i++) {
        for(int j = 0; j < edges.size(); j++) {
            if((isUsed[edges[j].begin] && !isUsed[edges[j].end]) || (!isUsed[edges[j].begin] && isUsed[edges[j].end])) {
                matrix[edges[j].begin][edges[j].end] = edges[j].weight;
                matrix[edges[j].end][edges[j].begin] = edges[j].weight;
                length += edges[j].weight;
                isUsed[edges[j].begin] = true;
                isUsed[edges[j].end] = true;
                continue;
            }
        }
    }
    cout << length << endl;
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < n; j++) {
            cout << matrix[i][j] << " ";
        }
        cout << endl;
    }
}


int main() {
    int matrixSize, edgesCount;
    cin >> matrixSize >> edgesCount;
    Edges edge;
    vector<Edges> edges;
    for (int i = 0; i < edgesCount; i++) {
            cin >> edge.begin >> edge.end >> edge.weight;
            edge.begin--;
            edge.end--;
            edges.push_back(edge);
    }
    sort(edges.begin(), edges.end(), comp);

    vector<vector<int>> matrix(matrixSize, vector<int>(matrixSize, 0));
    kraskalsAlg(edges, matrixSize, matrix);
}
