#include <iostream>
#include <vector>

using namespace std;

void printPath(int finalNode, vector<int> &previous, vector<int> &distance) {
    vector<int> path;
    int temp = finalNode;
    path.push_back(temp);
    while (previous[temp] != -1) {
        temp = previous[temp];
        path.insert(path.begin(), temp);
    }
    cout << distance[finalNode] + 1 << endl;
    for (int i = 0; i < path.size(); i++) {
        cout << path[i] + 1 << " ";
    }
}

void breadthFirstSearch(int startNode, const vector<vector<int>> &adjacencyMatrix, vector<bool> &isUsed, int graphSize,
                        int finalNode) {
    int temp;
    vector<int> previous(graphSize, -1);
    vector<int> distance(graphSize, -1);

    if (startNode == finalNode) {
        cout << "0";
        return;
    }

    vector<int> queue;
    queue.push_back(startNode);
    isUsed[startNode] = true;
    while (queue.size() > 0) {
        temp = queue[0];
        queue.erase(queue.begin());
        for (int j = 0; j < graphSize; j++) {
            if (!isUsed[j] && adjacencyMatrix[temp][j] != 0) {
                queue.push_back(j);
                isUsed[j] = true;
                distance[j] = distance[temp] + 1;
                previous[j] = temp;
            }
        }
    }

    if (distance[finalNode] == -1) {
        cout << distance[finalNode];
        return;
    } else {
        printPath(finalNode, previous, distance);
    }
}

int main() {
    int graphSize;
    cin >> graphSize;
    vector<vector<int>> adjacencyMatrix(graphSize, vector<int>(graphSize));
    vector<bool> isUsed(graphSize, false);
    for (int i = 0; i < graphSize; i++) {
        for (int j = 0; j < graphSize; j++) {
            cin >> adjacencyMatrix[i][j];
        }
    }
    int startNode, finalNode;
    cin >> startNode >> finalNode;

    breadthFirstSearch(startNode - 1, adjacencyMatrix, isUsed, graphSize, finalNode - 1);
}
