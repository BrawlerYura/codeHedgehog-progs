#include <iostream>

using namespace std;

int main() {
   int n;
   cin >> n;
   int **adjacencyMatrix = new int*[n];
   for(int i = 0; i < n; i++) {
       adjacencyMatrix[i] = new int[n];
   }
   for(int i = 0; i < n; i++) {
       for(int j = 0; j < n; j++) {
           cin >> adjacencyMatrix[i][j];
       }
   }
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < n; j++) {
            if(adjacencyMatrix[i][j] == 1) {
                if(i <= j) {
                    cout << i + 1 << " " << j + 1 << endl;
                }
            }
        }
    }
}
