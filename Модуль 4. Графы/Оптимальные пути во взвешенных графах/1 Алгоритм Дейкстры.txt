#include <iostream>
#include <vector>
#include <climits>

using namespace std;

void dijkstrasAlg(vector<vector<int>> &matrix, int n, int s, int f) {
    vector <int> distance(n, INT_MAX);
    vector <bool> isProcessed(n, false);
    distance[s] = 0;
    int minDistance = 0;
    int minVertex = s;

    while (minDistance < INT_MAX) {
        int temp = minVertex;
        isProcessed[temp] = true;
        for (int i = 0; i < n; i++)
            if (distance[i] > distance[temp] + matrix[temp][i] && matrix[temp][i] >= 0) {
                distance[i] = distance[temp] + matrix[temp][i];
            }
        minDistance = INT_MAX;
        for (int i = 0; i < n; i++) {
            if (!isProcessed[i] && distance[i] < minDistance) {
                minDistance = distance[i];
                minVertex = i;
            }
        }
    }
    if(distance[f] == INT_MAX) {
        cout << -1;
    }else {
        cout << distance[f];
    }
}

int main() {
    int matrixSize, start, finish;
    cin >> matrixSize >> start >> finish;
    vector<vector<int>> adjacencyMatrix(matrixSize, vector<int>(matrixSize));
    for (int i = 0; i < matrixSize; i++) {
        for (int j = 0; j < matrixSize; j++) {
            cin >> adjacencyMatrix[i][j];
        }
    }

    dijkstrasAlg(adjacencyMatrix, matrixSize, start - 1, finish - 1);
}
