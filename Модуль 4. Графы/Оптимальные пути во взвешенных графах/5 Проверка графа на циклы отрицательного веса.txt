#include <iostream>
#include <vector>
#include <climits>

using namespace std;

struct Edges {
    int begin, end, weight;
};

void printPath(vector<int> &previous, int flag, int n) {
    vector<int> path;
    int temp = flag;
    for (int i = 0; i < n; i++) {
        temp = previous[temp];
    }

    int current = temp;
    while (true) {
        path.insert(path.begin(), current);
        if (current == temp && path.size() >= 2) {
            break;
        }
        current = previous[current];
    }

    cout << "YES" << endl << path.size() << endl;
    for (int i = 0; i < path.size(); i++) {
        cout << path[i] + 1 << " ";
    }
}

void fordsAlg(vector<Edges> &edges, int n, int edgesCount, int s) {
    vector<int> distance(n, 100000);
    vector<int> previous(n, -1);
    distance[s] = 0;
    int flag = INT_MAX;
    for (int k = 0; k < n; k++) {
        flag = INT_MAX;
        for (int i = 0; i < edgesCount; i++) {
            if (distance[edges[i].begin] + edges[i].weight < distance[edges[i].end]) {
                distance[edges[i].end] = distance[edges[i].begin] + edges[i].weight;
                previous[edges[i].end] = edges[i].begin;
                flag = edges[i].end;
            }
        }
    }
    if (flag == INT_MAX) {
        cout << "NO";
        return;
    } else {
        printPath(previous, flag, n);
        return;
    }
}


int main() {
    int matrixSize;
    cin >> matrixSize;
    vector<vector<int>> adjacencyMatrix(matrixSize, vector<int>(matrixSize));
    vector<Edges> edges;
    Edges edge{};
    for (int i = 0; i < matrixSize; i++) {
        for (int j = 0; j < matrixSize; j++) {
            cin >> adjacencyMatrix[i][j];
            if (adjacencyMatrix[i][j] == 100000) {
                continue;
            }
            edge.begin = i;
            edge.end = j;
            edge.weight = adjacencyMatrix[i][j];
            edges.push_back(edge);
        }
    }

    fordsAlg(edges, matrixSize, edges.size(), 0);
}
