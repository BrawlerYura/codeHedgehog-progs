using System;
using System.Collections.Specialized;
using System.Globalization;
using System.Linq;
using System.Linq.Expressions;
using System.Text.RegularExpressions;
//Делали вместе с Ушаковым Федором
class Program
{
    static void Main()
    {
        string temp;
        string text = null;
        do
        {
            temp = Console.ReadLine();
            text += temp;
            text += " ";
        } while (temp != null);
        
        Regex regex = new Regex(@"\w{1,250}@[a-zA-Z0-9][a-zA-Z0-9-\.]*\.[a-z]{2,4}");
        MatchCollection matches = regex.Matches(text);

        for(int i = 0; i < matches.Count; i++)
        {
            if (matches[i].Length <= 255)
            {
                Regex repeatedСharacters = new Regex(@"(-{2,}|\.{2,}|\.-|-\.)");
                Match matchRepeatedСharacters = repeatedСharacters.Match(matches[i].ToString());

                if (!matchRepeatedСharacters.Success)
                {
                    Regex domainCount = new Regex(@"@.*");
                    Match matchDomain = domainCount.Match(matches[i].ToString());

                    if(matchDomain.Length <= 67 && matchDomain.Length >= 5)
                    {
                        Console.WriteLine(matches[i]);
                    }
                }
            }
        }
    }
}
