using System;
using System.Collections.Specialized;
using System.Globalization;
using System.Linq;
using System.Linq.Expressions;
using System.Text.RegularExpressions;
//Делали вместе с Ушаковым Федором
class Program
{
    static void Main()
    {
        string temp;
        string text = null;
        do
        {
            temp = Console.ReadLine();
            text += temp;
            text += " ";
        } while (temp != null);

        string date = @"((([1-2][0-9]|3[0-1]|0[1-9])\.(0[13-9]|1[0-2])|(0[13-9]|1[0-2])\.([1-2][0-9]|3[0-1]|0[1-9]))\.(\d{4}|\d{2}))";
        string dateWithFeb = @"(([1-2][0-9]|0[1-9])\.02|02\.([1-2][0-9]|0[1-9]))\.(\d{4}|\d{2})";
        Regex regex = new Regex($@"({date}|{dateWithFeb})");
        MatchCollection matches = regex.Matches(text);

        for(int i = 0; i < matches.Count; i++)
        {
            Console.WriteLine(matches[i]);
        }
    }
}
