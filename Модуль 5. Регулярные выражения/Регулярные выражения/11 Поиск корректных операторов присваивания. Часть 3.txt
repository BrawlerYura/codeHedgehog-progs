using System;
using System.Linq;
using System.Linq.Expressions;
using System.Text.RegularExpressions;
//Делали вместе с Ушаковым Федором
class Program
{
    static void Main()
    {
        string temp;
        string text = null;
        do
        {
            temp = Console.ReadLine();
            text += temp;
            text += " ";
        } while (temp != null);

        string nameOfVariable = @"[a-z][a-z0-9]*";
        string elementOfArray = $@"{nameOfVariable}(\[[a-z0-9]*\])?";

        string arithmeticExpression = $@"(\d+|{nameOfVariable}|{elementOfArray})\s*([+*/-]\s*(\d+|{nameOfVariable}|{elementOfArray})\s*)*";
        string elementOfArrayWithExpression = $@"[a-z][a-z0-9]*(\[{arithmeticExpression}\])?";
        string operand = $@"(\d+|{nameOfVariable}|{elementOfArrayWithExpression})";
        string ExpressionWithElementOfArray = $@"{operand}\s*([+*/-]\s*{operand}\s*)*";

        Regex regex = new Regex($@"({elementOfArrayWithExpression}|{nameOfVariable})\s*=\s*({nameOfVariable}|{elementOfArrayWithExpression}|{ExpressionWithElementOfArray});");
        MatchCollection matches = regex.Matches(text);
        for (int i = 0; i < matches.Count(); i++)
        {
            Console.WriteLine(matches[i]);
        }
    }
}
