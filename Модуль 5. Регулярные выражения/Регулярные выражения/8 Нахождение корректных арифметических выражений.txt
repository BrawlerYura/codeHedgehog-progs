using System;
using System.Linq;
using System.Text.RegularExpressions;
//Делали вместе с Ушаковым Федором
class Program
{
    static void Main()
    {
        string temp;
        string text = null;
        do
        {
            temp = Console.ReadLine();
            text += temp;
            text += " ";
        } while (temp != null);

        Regex regex = new Regex(@"[1-9]\d*(([+*-]\d+)|(/[1-9]\d*))*");
        MatchCollection matches = regex.Matches(text);

        for (int i = 0; i < matches.Count(); i++)
        {
            Console.WriteLine(matches[i].ToString());
        }
    }
}
