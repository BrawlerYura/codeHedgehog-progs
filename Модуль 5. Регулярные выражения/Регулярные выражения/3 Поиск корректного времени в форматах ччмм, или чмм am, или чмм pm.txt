using System;
using System.Linq;
using System.Text.RegularExpressions;

class Program {
    static void Main() {
        string temp;
        string text = null;
        do {
            temp = Console.ReadLine();
            text += temp;
            text += " ";
        } while (temp != null);

        Regex regex = new Regex(@"((1[0-2]|[0-9]):[0-5][0-9]\s(am|pm)|([0-1][0-9]|2[0-3]):[0-5][0-9])");
        MatchCollection matches = regex.Matches(text);
        
        for (int i = 0; i < matches.Count(); i++)
        {
            Console.WriteLine(matches[i].ToString());
        }
    }
}
