using System;
using System.Linq;
using System.Collections.Generic;
using System.Text.RegularExpressions;

class Program {
    static void Main() {
        string text = "a";
        int index = 0;
        while (text != null) {
            text = Console.ReadLine();
          if (text == null) {
            break;
          }
            Regex regex1 = new Regex(@"^Merlin speaks");
            MatchCollection matches = regex1.Matches(text);
            Regex regex2 = new Regex("Avada-ke-davra");
            Match match = regex2.Match(text);
            if (matches.Count == 1) {
                Console.WriteLine((index + 1) + " " + (match.Index + 1));
            }
            index++;
        }
    }
}
