using System;
using System.Collections.Generic;
using System.Collections.Specialized;
using System.Globalization;
using System.Linq;
using System.Linq.Expressions;
using System.Reflection.Emit;
using System.Text.RegularExpressions;
//Делали вместе с Кирсновым Александром
//Спасибо, папаша, за эту прекрасную статейку: https://www.codeproject.com/Articles/21183/In-Depth-with-NET-RegEx-Balanced-Grouping
class Program
{
    static void Main()
    {
        string temp;
        string text = null;
        do
        {
            temp = Console.ReadLine();
            text += temp;
            text += " ";
        } while (temp != null);

        string parenthesesSequence = @"?>\((?<LEVEL>)(?<CURRENT>)|(?=\))(?<LAST-CURRENT>)(?(?<=\(\k<LAST>)(?<-LEVEL>\)))";
        string curlyBracketsSequence = @"\{(?<LEVEL>)(?<CURRENT>)|(?=\})(?<LAST-CURRENT>)(?(?<=\{\k<LAST>)(?<-LEVEL>\}))";
        Regex regex = new Regex($@"({parenthesesSequence}|{curlyBracketsSequence})*(?(LEVEL)(?!))");
        MatchCollection matches = regex.Matches(text);

        for(int i = 0; i < matches.Count; i++)
        {
            Console.WriteLine(matches[i]);
        }
    }
}
