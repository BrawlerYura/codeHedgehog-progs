using System;
using System.Linq;
using System.Linq.Expressions;
using System.Text.RegularExpressions;
//Делали вместе с Ушаковым Федором
class Program
{
    static void Main()
    {
        string temp;
        string text = null;
        do
        {
            temp = Console.ReadLine();
            text += temp;
            text += " ";
        } while (temp != null);
        string variable = @"[a-z][a-z0-9]*(\[[a-z0-9]*\])?";
        string arithmeticExpression = $@"(\d+|{variable})\s*([+*/-]\s*\d+\s*|[+*/-]\s*{variable}\s*)*";
        Regex regex = new Regex($@"{variable}\s*=\s*({variable}|{arithmeticExpression});");
        MatchCollection matches = regex.Matches(text);
        for (int i = 0; i < matches.Count(); i++)
        {
            Console.WriteLine(matches[i]);
        }
    }
}
