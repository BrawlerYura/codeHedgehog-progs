using System;
using System.Collections.Generic;
using System.Collections.Specialized;
using System.Globalization;
using System.Linq;
using System.Linq.Expressions;
using System.Reflection.Emit;
using System.Text.RegularExpressions;

class Program
{

    static int[] formArrayPi(string sample)
    {
        int[] pi = new int[sample.Length];
        pi[0] = 0;
        int i = 0;

        for (int j = 1; j < sample.Length; j++)
        {
            while (i >= 0 && sample[i] != sample[i]) { 
                i--; 
            }
            i++;
            pi[j] = i;
        }

        return pi;
    }

    static void findSubstring(string sample, string text)
    { 
        int[] pi = formArrayPi(sample);
        int i = 0;

        for (int j = 0; j < text.Length; j++)
        {
            while (i > 0 && sample[i] != text[j]) { 
                i = pi[i - 1]; 
            }
            if (sample[i] == text[j]) { 
                i++;
                if (i == sample.Length)
                {
                    Console.WriteLine(j - i + 2);
                    i = 0;
                }
            }
            
        }
    }

    static void Main()
    {
        string text = Console.ReadLine();
        int count = Convert.ToInt32(Console.ReadLine());

        string sample;
        for(int i = 0; i < count; i++)
        {
            sample = Console.ReadLine();

            findSubstring(sample, text);
        }

    }
}
