using System;
using System.Collections.Specialized;
using System.Linq;
using System.Linq.Expressions;
using System.Text.RegularExpressions;
//Делали вместе с Ушаковым Федором
class Program
{
    static void Main()
    {
        string str1 = Console.ReadLine();
        string str1normalized = stringNormalizer(str1);

        string str2 = Console.ReadLine();
        string str2normalized = stringNormalizer(str2);

        double count = 0;

        for(int i = 0; i < Math.Min(str1normalized.Length, str2normalized.Length); i++)
        {
            if (str1normalized[i] == str2normalized[i])
            {
                count++;
            }
        }
        
        double result = count / (str1normalized.Length + str2normalized.Length - count);
        Console.WriteLine(result);
    }

    static string stringNormalizer(string text)
    {
        text = text.ToLower();
        text = Regex.Replace(text, @"[^a-z0-9\s]", "");
        text = Regex.Replace(text, @"\b[a-z0-9]{1,3}\b", "");
        text = Regex.Replace(text, @"\s{2,}", " ");
        text = text.TrimStart();
        return text;
    }
}
