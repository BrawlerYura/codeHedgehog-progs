using System;
using System.Collections.Specialized;
using System.Linq;
using System.Linq.Expressions;
using System.Text.RegularExpressions;
//Делали вместе с Ушаковым Федором
class Program
{
    static void Main()
    {
        string str1 = Console.ReadLine();
        string str1normalized = stringNormalizer(str1);

        string str2 = Console.ReadLine();
        string str2normalized = stringNormalizer(str2);

        int[,] matrix = new int[str1normalized.Length + 1, str2normalized.Length + 1];
        for (int i = 0; i < str1normalized.Length + 1; i++)
        {
            for (int j = 0; j < str2normalized.Length + 1; j++)
            {
                if (i == 0 && j == 0) {
                    matrix[i, j] = 0;
                } else if (j == 0 && i > 0) {
                    matrix[i, j] = i;
                } else if (i == 0 && j > 0) {
                    matrix[i, j] = j;
                } else
                {
                    matrix[i, j] = Math.Min(Math.Min(matrix[i, j - 1] + 1, matrix[i - 1, j] + 1), matrix[i - 1, j - 1] + (str1normalized[i - 1] == str2normalized[j - 1] ? 0 : 1));
                }
            }
        }
       

        Console.WriteLine(matrix[str1normalized.Length, str2normalized.Length]);
    }

    static string stringNormalizer(string text)
    {
        text = text.ToLower();
        text = Regex.Replace(text, @"[^a-z0-9\s]", "");
        text = Regex.Replace(text, @"\b[a-z0-9]{1,3}\b", "");
        text = Regex.Replace(text, @"\s{2,}", " ");
        text = text.TrimStart();
        return text;
    }
}
